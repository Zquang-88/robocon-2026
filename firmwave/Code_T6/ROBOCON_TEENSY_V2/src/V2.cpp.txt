// ===========================================
void setMotor(int pwm,
              int RPWM,
              int LPWM)
{
    pwm = constrain(pwm, -255, 255);
    
    if(pwm > 0)
    {
        analogWrite(RPWM, pwm);
        analogWrite(LPWM, 0);
    }
    else if(pwm < 0)
    {
        analogWrite(RPWM, 0);
        analogWrite(LPWM, -pwm);
    }
    else
    {
        analogWrite(RPWM, 0);
        analogWrite(LPWM, 0);
    }
}

// ===========================================
void updatePID(MotorPID &m)
{
    float error =
        m.targetSpeed -
        m.currentSpeed;

    // Integral
    m.integral += error;

    m.integral =
        constrain(m.integral,
                  -100,
                  100);

    // Derivative
    float derivative =
        error - m.lastError;

    m.lastError = error;

    // PID Output
    float output =
        m.kp * error +
        m.ki * m.integral +
        m.kd * derivative;

    // FeedForward
    int ff =
        abs(m.targetSpeed) * 5.65;

    // PWM
    if(m.targetSpeed > 0)
    {
        m.pwm =
            ff + output;
    }
    else if(m.targetSpeed < 0)
    {
        m.pwm =
            -ff + output;
    }
    else
    {
        m.pwm = 0;
        m.integral = 0;
    }

    m.pwm =
        constrain(m.pwm,
                  -255,
                  255);
}

// ===========================================
float readYaw()
{
    imu::Vector<3> euler =
        bno.getVector(
            Adafruit_BNO055::VECTOR_EULER);

    return euler.x();
}
float headingPID()
{       error = targetYaw - currentYaw;

    if(error > 180)
        error -= 360;

    if(error < -180)
        error += 360;

    headingIntegral += error;

    headingIntegral =
        constrain(
            headingIntegral,
            -50,
            50);

    float derivative = error - headingLastError;

    headingLastError = error;

    float output =
        headingKp * error +
        headingKi * headingIntegral +
        headingKd * derivative;

    output =
        constrain(
            output,
            -20,
            20);

    return output;
}

void moveForward(float mm)
{
    targetDistanceMM = mm;

    startFL = encFL.read();
    startFR = encFR.read();
    startBL = encBL.read();
    startBR = encBR.read();

    moveDone = false;
}

void updatePositionPID()
{
    long dFL = encFL.read() - startFL;
    long dFR = encFR.read() - startFR;
    long dBL = encBL.read() - startBL;
    long dBR = encBR.read() - startBR;


    switch(moveType)
    {
    case FORWARD:

        avgCount =
        (dFL + dFR + dBL + dBR)/4.0;

        break;

    case RIGHT:

        avgCount =
        (dFL - dFR - dBL + dBR)/4.0;

        break;

    case LEFT:

        avgCount =
        (-dFL + dFR + dBL - dBR)/4.0;

        break;
    }

    currentDistance =
        avgCount * mmPerCount;

    positionError =
        targetDistanceMM -
        currentDistance;

        speedCmd = positionKp * positionError;

        speedCmd =
            constrain(speedCmd,
                    -6,
                    6);

    // giảm tốc khi gần đích
    if(abs(positionError) < 100)
    {
        speedCmd =
            constrain(speedCmd, -2,2);
    }

    // tới đích
    if(abs(positionError) < 30)
    {
        speedCmd = 0;

        FL.targetSpeed = 0;
        FR.targetSpeed = 0;
        BL.targetSpeed = 0;
        BR.targetSpeed = 0;

        moveDone = true;
    }

}

//================HÀM ĐỌC CẢM BIẾN LINE 2 BÊN THÂN====================//
void updateLineCounter()
{
    bool leftNow =
        digitalRead(LINE_LEFT_PIN);

    bool rightNow =
        digitalRead(LINE_RIGHT_PIN);

    // đen = LOW
    if(leftNow == HIGH &&
    lastLineLeft == LOW &&
    leftDebounce > 100)
    {
        leftLineCount++;
        leftDebounce = 0;


        Serial.print("LEFT=");
        Serial.println(leftLineCount);
    }

   if(rightNow == HIGH &&
   lastLineRight == LOW &&
   rightDebounce > 100)
{
    rightLineCount++;

    rightDebounce = 0;

    Serial.print("RIGHT=");
    Serial.println(rightLineCount);
}
lastLineLeft = leftNow;
lastLineRight = rightNow;
}

//================UART ESP32====================//
void readESP()
{
    while(Serial2.available())
    {
        char c =
            Serial2.read();

        if(c == '\n')
        {
            espMsg.trim();

            if(espMsg == "DONE")
            {
                espDone = true;
            }

            espMsg = "";
        }
        else
        {
            espMsg += c;
        }
    }
}

int readTOFFiltered()
{
    tofBuffer[tofIndex] = tof.read();

    tofIndex++;

    if(tofIndex >= TOF_FILTER_SIZE)
        tofIndex = 0;

    long sum = 0;

    for(int i=0;i<TOF_FILTER_SIZE;i++)
    {
        sum += tofBuffer[i];
    }

    return sum / TOF_FILTER_SIZE;
}
    void updateTOFControl()
    {
        int dist = readTOFFiltered();

        tofError =
            targetTOF - dist;

        speedCmd =
            tofError * tofKp;

        speedCmd =
            constrain(
                speedCmd,
                -2,
                2);
    }
