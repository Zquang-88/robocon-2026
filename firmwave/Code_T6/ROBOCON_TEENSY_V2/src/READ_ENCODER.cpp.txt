#include <Encoder.h>

//================ ENCODER ==================
Encoder encFL(18, 19);
Encoder encFR(22, 23);
Encoder encBL(15, 14);
Encoder encBR(29, 28);

//================ MOTOR ====================
#define FL_RPWM 36
#define FL_LPWM 37

#define FR_RPWM 4
#define FR_LPWM 5

#define BL_RPWM 25
#define BL_LPWM 24

#define BR_RPWM 3
#define BR_LPWM 2

//===========================================

void setMotor(int rpwmPin, int lpwmPin, int pwm)
{
    pwm = constrain(pwm, -255, 255);

    if(pwm > 0)
    {
        analogWrite(rpwmPin, pwm);
        analogWrite(lpwmPin, 0);
    }
    else if(pwm < 0)
    {
        analogWrite(rpwmPin, 0);
        analogWrite(lpwmPin, -pwm);
    }
    else
    {
        analogWrite(rpwmPin, 0);
        analogWrite(lpwmPin, 0);
    }
}

void stopAll()
{
    setMotor(FL_RPWM, FL_LPWM, 0);
    setMotor(FR_RPWM, FR_LPWM, 0);
    setMotor(BL_RPWM, BL_LPWM, 0);
    setMotor(BR_RPWM, BR_LPWM, 0);
}

void setup()
{
    Serial.begin(115200);

    pinMode(FL_RPWM, OUTPUT);
    pinMode(FL_LPWM, OUTPUT);

    pinMode(FR_RPWM, OUTPUT);
    pinMode(FR_LPWM, OUTPUT);

    pinMode(BL_RPWM, OUTPUT);
    pinMode(BL_LPWM, OUTPUT);

    pinMode(BR_RPWM, OUTPUT);
    pinMode(BR_LPWM, OUTPUT);

    analogWriteResolution(8);

    stopAll();

    delay(1000);

    Serial.println("START");
}

void loop()
{
    // Chạy tất cả động cơ
    setMotor(FL_RPWM, FL_LPWM, 120);
    setMotor(FR_RPWM, FR_LPWM, 120);
    setMotor(BL_RPWM, BL_LPWM, 120);
    setMotor(BR_RPWM, BR_LPWM, 120);

    static uint32_t t = 0;

    if(millis() - t > 200)
    {
        t = millis();

        long fl = encFL.read();
        long fr = encFR.read();
        long bl = encBL.read();
        long br = encBR.read();

        Serial.print("FL=");
        Serial.print(fl);

        Serial.print("\tFR=");
        Serial.print(fr);

        Serial.print("\tBL=");
        Serial.print(bl);

        Serial.print("\tBR=");
        Serial.println(br);
    }
}
