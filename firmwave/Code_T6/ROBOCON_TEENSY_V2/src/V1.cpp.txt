#define FL_RPWM 36
#define FL_LPWM 37

#define FR_LPWM 5

#define BL_RPWM 25
#define BL_LPWM 24

#define BR_RPWM 3
#define BR_LPWM 2

// ===========================================
struct MotorPID
{
    float kp;
    float ki;
    float kd;

    int feedForward;

    float targetSpeed;
    float currentSpeed;

    float integral;
    float lastError;

    int pwm;
};

MotorPID FL;
MotorPID FR;
MotorPID BL;
MotorPID BR;

// ===========================================
long lastFL = 0;
long lastFR = 0;
long lastBL = 0;
long lastBR = 0;

elapsedMillis pidTimer;
elapsedMillis imuTimer;
// ===========================================
void setMotor(int pwm,
              int RPWM,
              int LPWM)
{
    pwm = constrain(pwm, -255, 255);

    if(pwm > 0)
    {
        analogWrite(RPWM, pwm);
        analogWrite(LPWM, 0);
    }
    else if(pwm < 0)
    {
        analogWrite(RPWM, 0);
        analogWrite(LPWM, -pwm);
    }
    else
    {
        analogWrite(RPWM, 0);
        analogWrite(LPWM, 0);
    }
}

// ===========================================
void updatePID(MotorPID &m)
{
    float error =
        m.targetSpeed -
        m.currentSpeed;

    // Integral
    m.integral += error;

    m.integral =
        constrain(m.integral,
                  -100,
                  100);

    // Derivative
    float derivative =
        error - m.lastError;

    m.lastError = error;

    // PID Output
    float output =
        m.kp * error +
        m.ki * m.integral +
        m.kd * derivative;

    // FeedForward
    int ff =
        abs(m.targetSpeed) * 5.65;

    // PWM
    if(m.targetSpeed > 0)
    {
        m.pwm =
            ff + output;
    }
    else if(m.targetSpeed < 0)
    {
        m.pwm =
            -ff + output;
    }
    else
    {
        m.pwm = 0;
        m.integral = 0;
    }

    m.pwm =
        constrain(m.pwm,
                  -255,
                  255);
}
// ===========================================
void setupPID()
{
    // ===== FL =====
    FL.kp = 3.0;
    FL.ki = 0.28;
    FL.kd = 0.0;

    // ===== FR =====
    FR.kp = 3.0;
    FR.ki = 0.28;
    FR.kd = 0.02;

    // ===== BL =====
    BL.kp = 3.0;
    BL.ki = 0.29;
    BL.kd = 0.02;

    // ===== BR =====
    BR.kp = 3.0;
    BR.ki = 0.29;
    BR.kd = 0.02;

}

// ===========================================
float readYaw()
{
    imu::Vector<3> euler =
        bno.getVector(
            Adafruit_BNO055::VECTOR_EULER);

    return euler.x();
}
float headingPID()
{       error =
        targetYaw -
        currentYaw;

    if(error > 180)
        error -= 360;

    if(error < -180)
        error += 360;

    headingIntegral += error;

    headingIntegral =
        constrain(
            headingIntegral,
            -50,
            50);

    float derivative =
        error -
        headingLastError;

    headingLastError =
        error;

    float output =
        headingKp * error +
        headingKi * headingIntegral +
        headingKd * derivative;

    output =
        constrain(
            output,
            -20,
            20);

    return output;
}


//===================================================
void moveForward(float mm)
{
    targetDistanceMM = mm;

    startFL = encFL.read();
    startFR = encFR.read();
    startBL = encBL.read();
    startBR = encBR.read();

    moveDone = false;
}

void updatePositionPID()
{
    long dFL = encFL.read() - startFL;
    long dFR = encFR.read() - startFR;
    long dBL = encBL.read() - startBL;
    long dBR = encBR.read() - startBR;

    float avgCount =
        (dFL + dFR + dBL + dBR) / 4.0;

    currentDistance =
        avgCount * mmPerCount;

    float error =
        targetDistanceMM -
        currentDistance;

    speedCmd =
        positionKp * error;

    speedCmd =
        constrain(speedCmd,
                  -10,
                   10);

    if(abs(error) < 100)
    {
        speedCmd =
            constrain(speedCmd,
                      -5,
                       5);
    }

    if(abs(error) < 5)
    {
        speedCmd = 0;
        moveDone = true;
    }
}

{
    Serial.begin(115200);

    pinMode(FL_RPWM, OUTPUT);
    pinMode(FL_LPWM, OUTPUT);

    pinMode(FR_RPWM, OUTPUT);
    pinMode(FR_LPWM, OUTPUT);

    pinMode(BL_LPWM, OUTPUT);

    pinMode(BR_RPWM, OUTPUT);
    pinMode(BR_LPWM, OUTPUT);

    analogWriteResolution(8);

    analogWriteFrequency(FL_RPWM, 20000);
    analogWriteFrequency(FL_LPWM, 20000);

    analogWriteFrequency(FR_RPWM, 20000);
    analogWriteFrequency(FR_LPWM, 20000);

    analogWriteFrequency(BL_RPWM, 20000);
    analogWriteFrequency(BL_LPWM, 20000);

    analogWriteFrequency(BR_RPWM, 20000);
    analogWriteFrequency(BR_LPWM, 20000);

    setupPID();

    // test
    float TOCDO = 6;
    FL.targetSpeed = TOCDO;
    FR.targetSpeed = TOCDO;
    BL.targetSpeed = TOCDO;
    BR.targetSpeed = TOCDO;

    //YAW
    delay(1000);

Wire1.begin();

{
    while(1);
}

bno.setExtCrystalUse(true);

//=================Position============

}

void loop()
{
    // =====================================
    // IMU + Heading PID (100Hz)
    // =====================================
    if(imuTimer >= 10)
    {
        imuTimer = 0;

        currentYaw = readYaw();

        omega = headingPID();

        lastYaw = currentYaw;

    // PID tốc độ + PID vị trí (100Hz)
    // =====================================
