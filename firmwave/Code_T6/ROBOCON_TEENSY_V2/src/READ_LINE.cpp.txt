void setup() { // Khoi tao Serial, I2C, cam bien line, PCF8574 va nap calib ban dau.
    Serial.begin(115200);
    while (!Serial && millis() < 3000);

    Wire1.begin();

    Serial.println();
    Serial.println("======================================");
    Serial.println("TEENSY 4.1 - MCP3008 LINE SENSOR TEST");
    Serial.println("MCP3008 CS = 10");
    Serial.println("PCF8574 I2C = 0x20 tren Wire1");
    Serial.println("======================================");

    Serial.println("Init LineSensor...");
    line.begin();

    // Theo gia tri ban test: den 900, trang 100
    line.setBlackLineMode(true);

    // Loc ADC trung binh 3 mau
    line.setReadSamples(3);

    // Chong nhay bit
    line.setMargin(30);

    if (line.loadFromEEPROM(0)) {
        Serial.println("Load EEPROM calib OK");
    }
    else {
        Serial.println("EEPROM chua co calib, dung mac dinh:");
        Serial.println("White = 100, Black = 900, Threshold = 500");
    }

    line.printCalib();

    Serial.println("Init PCF8574...");
    if (pcf.begin()) {
        pcfReady = true;
        Serial.println("PCF8574 OK");
    }
    else {
        pcfReady = false;
        Serial.println("PCF8574 not found. Van test duoc bang Serial.");
    }

    printHelp();

    lastPrint = millis();
}

// ================= LOOP =================
void loop() { // Vong lap chinh: doc Serial, doc nut va in du lieu line theo chu ky.
    readSerial();
    readButton();

    uint32_t now = millis();

    if (now - lastPrint >= PRINT_MS) {
        lastPrint = now;
        printLineData();
    }
}
