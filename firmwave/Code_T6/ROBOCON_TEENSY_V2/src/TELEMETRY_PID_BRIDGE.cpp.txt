#include <Arduino.h>

// SiK/3DR air telemetry nối trực tiếp Teensy 4.1:
// Telemetry TX -> RX5 pin 34, Telemetry RX -> TX5 pin 33, GND -> GND.
#define TELEMETRY Serial5

struct PIDGains { float kp, ki, kd; };
PIDGains wheelPid[4], headingPid, linePid;
bool emergencyStop = false;

int wheelIndex(const String &id) {
  if (id == "FL") return 0; if (id == "FR") return 1;
  if (id == "RL") return 2; if (id == "RR") return 3;
  return -1;
}

void sendAck(const String &message) { TELEMETRY.println("ACK," + message); }
void sendError(const String &message) { TELEMETRY.println("ERR," + message); }

void parseCommand(String packet) {
  packet.trim();
  if (packet == "ESTOP") {
    emergencyStop = true;
    // stopAllMotors();
    sendAck("ESTOP");
    return;
  }
  if (packet == "RUN") { emergencyStop = false; sendAck("RUN"); return; }

  char buffer[128];
  packet.toCharArray(buffer, sizeof(buffer));
  char *cmd = strtok(buffer, ",");
  char *scope = strtok(nullptr, ",");
  char *target = strtok(nullptr, ",");
  char *kpText = strtok(nullptr, ",");
  char *kiText = strtok(nullptr, ",");
  char *kdText = strtok(nullptr, ",");
  if (!cmd || strcmp(cmd, "PID") || !scope || !target || !kpText || !kiText || !kdText) {
    sendError("BAD_PACKET"); return;
  }

  PIDGains gains = { atof(kpText), atof(kiText), atof(kdText) };
  if (!strcmp(scope, "WHEEL")) {
    if (!strcmp(target, "ALL")) for (auto &pid : wheelPid) pid = gains;
    else {
      int index = wheelIndex(target);
      if (index < 0) { sendError("BAD_WHEEL"); return; }
      wheelPid[index] = gains;
    }
  } else if (!strcmp(scope, "HEADING")) headingPid = gains;
  else if (!strcmp(scope, "LINE")) linePid = gains;
  else { sendError("BAD_SCOPE"); return; }

  sendAck(String("PID,") + scope + "," + target);
}

void telemetrySetup() {
  TELEMETRY.begin(57600, SERIAL_8N1);
  TELEMETRY.setTimeout(5);
}

void telemetryLoop(float rpmFL, float rpmFR, float rpmRL, float rpmRR,
                   float headingDeg, float lineError, float batteryV) {
  while (TELEMETRY.available()) parseCommand(TELEMETRY.readStringUntil('\n'));

  static uint32_t lastSend = 0;
  if (millis() - lastSend >= 10) {
    lastSend = millis();
    TELEMETRY.printf("TEL,%lu,%.1f,%.1f,%.1f,%.1f,%.2f,%.2f,%.2f\n",
      millis(), rpmFL, rpmFR, rpmRL, rpmRR, headingDeg, lineError, batteryV);
  }
}
