void setMotor(int pwm,
              int RPWM,
              int LPWM)
{
    pwm = constrain(pwm, -255, 255);
    
    if(pwm > 0)
    {
        analogWrite(RPWM, pwm);
        analogWrite(LPWM, 0);
    }
    else if(pwm < 0)
    {
        analogWrite(RPWM, 0);
        analogWrite(LPWM, -pwm);
    }
    else
    {
        analogWrite(RPWM, 0);
        analogWrite(LPWM, 0);
    }
}

// ===========================================
void updatePID(MotorPID &m)
{
    float error =
        m.targetSpeed -
        m.currentSpeed;

    // Integral
    m.integral += error;

    m.integral =
        constrain(m.integral,
                  -100,
                  100);

    // Derivative
    float derivative =
        error - m.lastError;

    m.lastError = error;

    // PID Output
    float output =
        m.kp * error +
        m.ki * m.integral +
        m.kd * derivative;

    // FeedForward
    int ff =
        abs(m.targetSpeed) * 5.65;

    // PWM
    if(m.targetSpeed > 0)
    {
        m.pwm =
            ff + output;
    }
    else if(m.targetSpeed < 0)
    {
        m.pwm =
            -ff + output;
    }
    else
    {
        m.pwm = 0;
        m.integral = 0;
    }

    m.pwm =
        constrain(m.pwm,
                  -255,
                  255);
}

uint8_t getLineBits()
{
    uint8_t bits = 0;

}

const int weight[8] =
{
    -350,
    -250,
    -150,
    -50,
     50,
    150,
    250,
    350
};
int getLineError()
{
    readLineArray();

    long sum = 0;
    long total = 0;

    for(int i=0;i<8;i++)
    {
        if(lineADC[i] > LINE_TH)
        {
            sum += weight[i];
            total++;
        }
    }

    if(total == 0)
        return 9999;

    return sum / total;
}

float readYaw()
{

}
float headingPID()
{       error = targetYaw - currentYaw;

    if(error > 180)
        error -= 360;

    if(error < -180)
        error += 360;

    headingIntegral += error;

    headingIntegral =
        constrain(
            headingIntegral,
            -50,
            50);

    float derivative = error - headingLastError;

    headingLastError = error;

    float output =
        headingKp * error +
        headingKi * headingIntegral +
        headingKd * derivative;

    output =
        constrain(
            output,
            -20,
            20);

    return output;
}

void moveForward(float mm)
{
    targetDistanceMM = mm;

    startFL = encFL.read();
    startFR = encFR.read();
    startBL = encBL.read();
    startBR = encBR.read();

    moveDone = false;
}

void updatePositionPID()
{
    long dFL = encFL.read() - startFL;
    long dFR = encFR.read() - startFR;
    long dBL = encBL.read() - startBL;
    long dBR = encBR.read() - startBR;


    switch(moveType)
    {
    case FORWARD:

        avgCount =
        (dFL + dFR + dBL + dBR)/4.0;

        break;

    case RIGHT:

        avgCount =
        (dFL - dFR - dBL + dBR)/4.0;

        break;

    case LEFT:

        avgCount =
        (-dFL + dFR + dBL - dBR)/4.0;

        break;
    }

    currentDistance =
        avgCount * mmPerCount;

    positionError =
        targetDistanceMM -
        currentDistance;

        speedCmd = positionKp * positionError;

        speedCmd =
            constrain(speedCmd,
                    -8,
                    8);

    // giảm tốc khi gần đích
    if(abs(positionError) < 150)
    {
        speedCmd =
            constrain(speedCmd, -5,5);
    }

    // tới đích
    if(abs(positionError) < 30)
    {
        speedCmd = 0;

        FL.targetSpeed = 0;
        FR.targetSpeed = 0;
        BL.targetSpeed = 0;
        BR.targetSpeed = 0;

        moveDone = true;
    }

}

void updateLineCounter()
{
    bool leftNow =
        digitalRead(LINE_LEFT_PIN);

    bool rightNow =
        digitalRead(LINE_RIGHT_PIN);

    // đen = LOW
    if(leftNow == HIGH &&
       lastLineLeft == LOW &&
       leftDebounce > 200)
    {
        leftLineCount++;

        leftDebounce = 0;

        Serial.print("LEFT=");
        Serial.println(leftLineCount);
    }

    if(rightNow == HIGH &&
       lastLineRight == LOW &&
       rightDebounce > 200)
    {
        rightLineCount++;

        rightDebounce = 0;

        Serial.print("RIGHT=");
        Serial.println(rightLineCount);
    }

    lastLineLeft = leftNow;
    lastLineRight = rightNow;
}

int readTOFFiltered()

    tofBuffer[tofIndex] = tof.read();

    tofIndex = 0;

  uint16_t readMCP(uint8_t channel)
{
    SPI.beginTransaction(
        SPISettings(
            1000000,
            MSBFIRST,
            SPI_MODE0));

    digitalWrite(MCP_CS, LOW);

    SPI.transfer(0x01);

    uint8_t high =
        SPI.transfer((0x08 | channel) << 4);

    uint8_t low =
        SPI.transfer(0);

    digitalWrite(MCP_CS, HIGH);

    SPI.endTransaction();

    return ((high & 0x03) << 8) | low;
}

bool centerOnLine()
{
    int error = getLineError();

    if(error == 9999)
        return false;

    // Đã vào giữa line
    if(abs(error) < 15)
    {
        stopRobot();
        return true;
    }

    int pwm;

    if(abs(error) > 250)
        pwm = 8;
    else if(abs(error) > 150)
        pwm = 6;
    else if(abs(error) > 80)
        pwm = 4;
    else
        pwm = 2;

    // Chỉnh chiều tại đây
    if(error > 0)
    {
        moveType = RIGHT;
        speedCmd = pwm;
    }
    else
    {
        moveType = LEFT;
        speedCmd = pwm;
    }

    return false;
}

void followLineForward()
{
    int error = getLineError();

    float lineCorrection = 0;

    if(error != 9999)
    {
        lineCorrection =
            error * lineFollowKp;

        lineCorrection =
            constrain(
                lineCorrection,
                -2.0,
                 2.0);
    }

    float vx = 3.0;
    float vy = 0;

    // line bên phải -> kéo trái
    if(error > 0)
        vy = -lineCorrection;

    // line bên trái -> kéo phải
    else
        vy = -lineCorrection;

    FL.targetSpeed =
        vx - vy + omega;

    FR.targetSpeed =
        vx + vy - omega;

    BL.targetSpeed =
        vx + vy + omega;

    BR.targetSpeed =
        vx - vy - omega;
}
