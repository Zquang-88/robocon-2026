#include "PCFInput.h"

PCFInput::PCFInput(TwoWire &wire, uint8_t address) {
    _wire = &wire;
    _address = address;

    _rawData = 0xFF;
    _ok = false;

    _debounceMs = 40;

    for (uint8_t i = 0; i < NUM_SW; i++) {
        _swStable[i] = false;
        _swLastStable[i] = false;
        _swReading[i] = false;
        _swPressed[i] = false;
        _swReleased[i] = false;
        _lastDebounceTime[i] = 0;
    }
}

bool PCFInput::begin(uint32_t clock) {
    _wire->begin();
    _wire->setClock(clock);

    // Ghi HIGH toàn bộ chân để PCF8574 đọc input
    _wire->beginTransmission(_address);
    _wire->write(0xFF);
    uint8_t error = _wire->endTransmission();

    _ok = (error == 0);

    // Đọc lần đầu để cập nhật raw
    if (_ok) {
        readRaw();

        bool initSW[NUM_SW] = {
            activeLow(4),
            activeLow(5),
            activeLow(6),
            activeLow(7)
        };

        for (uint8_t i = 0; i < NUM_SW; i++) {
            _swStable[i] = initSW[i];
            _swLastStable[i] = initSW[i];
            _swReading[i] = initSW[i];
        }
    }

    return _ok;
}

bool PCFInput::readRaw() {
    uint8_t n = _wire->requestFrom(_address, (uint8_t)1);

    if (n == 1 && _wire->available()) {
        _rawData = _wire->read();
        _ok = true;
        return true;
    }

    _ok = false;
    return false;
}

bool PCFInput::update() {
    // Reset event mỗi vòng update
    for (uint8_t i = 0; i < NUM_SW; i++) {
        _swPressed[i] = false;
        _swReleased[i] = false;
    }

    if (!readRaw()) {
        return false;
    }

    // P4 P5 P6 P7 là SW1 SW2 SW3 SW4
    updateButton(0, activeLow(5));
    updateButton(1, activeLow(4));
    updateButton(2, activeLow(7));
    updateButton(3, activeLow(6));

    return true;
}

void PCFInput::updateButton(uint8_t index, bool readingNow) {
    uint32_t now = millis();

    if (readingNow != _swReading[index]) {
        _swReading[index] = readingNow;
        _lastDebounceTime[index] = now;
    }

    if ((now - _lastDebounceTime[index]) >= _debounceMs) {
        if (_swStable[index] != _swReading[index]) {
            _swLastStable[index] = _swStable[index];
            _swStable[index] = _swReading[index];

            if (_swStable[index] == true && _swLastStable[index] == false) {
                _swPressed[index] = true;
            }

            if (_swStable[index] == false && _swLastStable[index] == true) {
                _swReleased[index] = true;
            }
        }
    }
}

bool PCFInput::activeLow(uint8_t bit) const {
    return !(_rawData & (1 << bit));
}

// DIP theo sơ đồ:
// P0 -> DIP4
// P1 -> DIP3
// P2 -> DIP2
// P3 -> DIP1

bool PCFInput::dip1() const {
    return activeLow(0);
}

bool PCFInput::dip2() const {
    return activeLow(1);
}

bool PCFInput::dip3() const {
    return activeLow(2 );
}

bool PCFInput::dip4() const {
    return activeLow(3);
}

uint8_t PCFInput::mode() const {
    uint8_t value = 0;

    value |= dip1() << 0;
    value |= dip2() << 1;
    value |= dip3() << 2;
    value |= dip4() << 3;

    return value;
}

// SW

bool PCFInput::getSW(uint8_t index) const {
    if (index >= NUM_SW) return false;
    return _swStable[index];
}

bool PCFInput::getSWPressed(uint8_t index) const {
    if (index >= NUM_SW) return false;
    return _swPressed[index];
}

bool PCFInput::getSWReleased(uint8_t index) const {
    if (index >= NUM_SW) return false;
    return _swReleased[index];
}

bool PCFInput::sw1() const {
    return getSW(0);
}

bool PCFInput::sw2() const {
    return getSW(1);
}

bool PCFInput::sw3() const {
    return getSW(2);
}

bool PCFInput::sw4() const {
    return getSW(3);
}

bool PCFInput::sw1Pressed() const {
    return getSWPressed(0);
}

bool PCFInput::sw2Pressed() const {
    return getSWPressed(1);
}

bool PCFInput::sw3Pressed() const {
    return getSWPressed(2);
}

bool PCFInput::sw4Pressed() const {
    return getSWPressed(3);
}

bool PCFInput::sw1Released() const {
    return getSWReleased(0);
}

bool PCFInput::sw2Released() const {
    return getSWReleased(1);
}

bool PCFInput::sw3Released() const {
    return getSWReleased(2);
}

bool PCFInput::sw4Released() const {
    return getSWReleased(3);
}

uint8_t PCFInput::raw() const {
    return _rawData;
}

bool PCFInput::ok() const {
    return _ok;
}
