#ifndef LINE_SENSOR_H
#define LINE_SENSOR_H

#include <Arduino.h>
#include <SPI.h>
#include <EEPROM.h>

#define LINE_LOST_ERROR 9999

class LineSensor {
public:
    LineSensor(uint8_t csPin);

    void begin();

    uint16_t readMCP3008(uint8_t channel);
    void readRaw();

    void learnWhite(uint16_t samples = 50);
    void learnBlack(uint16_t samples = 50);
    void updateThreshold();

    uint8_t readBits();

    bool saveToEEPROM(int address = 0);
    bool loadFromEEPROM(int address = 0);
    void resetCalibration();

    void printRaw();
    void printBits(uint8_t bits);
    void printCalib();

    uint16_t getADC(uint8_t ch);
    uint16_t getThreshold(uint8_t ch);

    void setBlackLineMode(bool mode);
    bool getBlackLineMode();

    void setReadSamples(uint8_t samples);
    void setMargin(uint16_t margin);

    int16_t getLineError();
    uint8_t countLine(uint8_t bits);
    bool isCenter(uint8_t bits);
    bool isLost(uint8_t bits);
    bool isCrossLine(uint8_t bits);

private:
    uint8_t _csPin;

    SPISettings mcpSPI = SPISettings(1000000, MSBFIRST, SPI_MODE0);

    uint16_t adcValue[8];
    uint16_t whiteValue[8];
    uint16_t blackValue[8];
    uint16_t thresholdValue[8];

    bool blackLineMode = true;

    uint8_t lastBits = 0;

    uint8_t readSamples = 3;
    uint16_t lineMargin = 30;

    bool isLineAtChannel(uint8_t ch, bool lastState);
};

#endif
